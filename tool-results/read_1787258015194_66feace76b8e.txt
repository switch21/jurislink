   168→   168→function fmtDate(d: string | null | undefined) {
   169→   169→  if (!d) return '—'
   170→   170→  try { return format(parseISO(d), 'dd/MM/yyyy', { locale: fr }) } catch { return '—' }
   171→   171→}
   172→   172→function fmtMoney(amount: number, code: string = 'XAF') {
   173→   173→  return new Intl.NumberFormat('fr-FR', { style: 'currency', currency: code, minimumFractionDigits: 0 }).format(amount)
   174→   174→}
   175→   175→function fmtFileSize(bytes: number) {
   176→   176→  if (bytes < 1024) return bytes + ' o'
   177→   177→  if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' Ko'
   178→   178→  return (bytes / 1048576).toFixed(1) + ' Mo'
   179→   179→}
   180→   180→function initials(name: string) {
