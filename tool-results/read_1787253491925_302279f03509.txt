   167→// ==================== Helpers ====================
   168→function fmtDate(d: string | null | undefined) {
   169→  if (!d) return '—'
   170→  try { return format(parseISO(d), 'dd/MM/yyyy', { locale: fr }) } catch { return '—' }
   171→}
   172→function fmtMoney(amount: number, code: string = 'XAF') {
   173→  return new Intl.NumberFormat('fr-FR', { style: 'currency', currency: code, minimumFractionDigits: 0 }).format(amount)
   174→}
   175→function fmtFileSize(bytes: number) {
   176→  if (bytes < 1024) return bytes + ' o'
   177→  if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' Ko'
   178→  return (bytes / 1048576).toFixed(1) + ' Mo'
   179→}
   180→function initials(name: string) {
   181→  return name.split(' ').map(n => n[0]).join('').toUpperCase().slice(0, 2)

   224→    e.preventDefault()

   228→      const res = await fetch('/api/auth/login', {
