   155→// ==================== Helpers ====================
   156→function fmtDate(d: string | null | undefined) {
   157→  if (!d) return '—'
   158→  try { return format(parseISO(d), 'dd/MM/yyyy', { locale: fr }) } catch { return '—' }
   159→}
   160→function fmtMoney(amount: number, code: string = 'XAF') {
   161→  return new Intl.NumberFormat('fr-FR', { style: 'currency', currency: code, minimumFractionDigits: 0 }).format(amount)
   162→}
   163→function fmtFileSize(bytes: number) {
   164→  if (bytes < 1024) return bytes + ' o'
   165→  if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' Ko'
   166→  return (bytes / 1048576).toFixed(1) + ' Mo'
   167→}
   168→function initials(name: string) {
   169→  return name.split(' ').map(n => n[0]).join('').toUpperCase().slice(0, 2)
   170→}

   172→// ==================== Login Page ====================
   173→function LoginPage() {
   174→  const { login } = useAppStore()
   175→  const [email, setEmail] = useState('')
   176→  const [password, setPassword] = useState('')
   177→  const [loading, setLoading] = useState(false)
   178→
   179→  const handleSubmit = async (e: React.FormEvent) => {
   180→    e.preventDefault()
   181→    if (!email || !password) { toast.error('Veuillez remplir tous les champs'); return }
   182→    setLoading(true)
   183→    try {
   184→      const res = await fetch('/api/auth/login', {
   185→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   186→        body: JSON.stringify({ email, password })
   187→      })
   188→      const data = await res.json()
   189→      if (!res.ok) { toast.error(data.error || 'Erreur de connexion'); return }
   190→      login(data)
   191→      toast.success(`Bienvenue, ${data.name} !`)
   192→    } catch {
   193→      toast.error('Erreur de connexion au serveur')
   194→    } finally { setLoading(false) }
   195→  }
   196→
   197→  return (
   198→    <div className="min-h-screen flex items-center justify-center bg-slate-100 p-4">
   199→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="w-full max-w-md">
   200→        <Card className="shadow-xl border-slate-200">
   201→          <CardHeader className="text-center pb-2">
   202→            <div className="mx-auto mb-4 flex items-center justify-center gap-2">
   203→              <span className="text-3xl text-amber-600">⬡</span>
   204→              <span className="text-2xl font-bold text-slate-900 tracking-tight">JurisLink</span>
   205→            </div>
   206→            <CardTitle className="text-xl text-slate-800">Connexion</CardTitle>
   207→            <CardDescription>Gestion juridique intelligente</CardDescription>
   208→          </CardHeader>
   209→          <CardContent>
   210→            <form onSubmit={handleSubmit} className="space-y-4">
   211→              <div className="space-y-2">
   212→                <Label htmlFor="email">Adresse e-mail</Label>
   213→                <Input id="email" type="email" placeholder="email@jurislink.com" value={email} onChange={e => setEmail(e.target.value)} />
   214→              </div>
   215→              <div className="space-y-2">
   216→                <Label htmlFor="password">Mot de passe</Label>
   217→                <Input id="password" type="password" placeholder="••••••••" value={password} onChange={e => setPassword(e.target.value)} />
   218→              </div>
   219→              <Button type="submit" className="w-full bg-slate-900 hover:bg-slate-800" disabled={loading}>
   220→                {loading ? <RefreshCw className="size-4 animate-spin" /> : 'Se connecter'}
   221→              </Button>
   222→            </form>
   223→          </CardContent>
   224→          <CardFooter className="justify-center">
   225→            <p className="text-xs text-slate-400">Démo : ngassa@jurislink.com / Admin@123</p>
   226→          </CardFooter>
   227→        </Card>
   228→      </motion.div>
   229→    </div>
   230→  )
   231→}

   307→  const { data: messages } = useQuery<Message[]>({

   312→  const unreadMsgCount = messages?.filter(m => !m.isRead && m.receiverId === user?.id).length || 0
   313→
   314→  const markAllRead = async () => {
   315→    if (!user?.tenantId || !user?.id) return

   317→      method: 'POST', headers: { 'Content-Type': 'application/json' },
   318→      body: JSON.stringify({ tenantId: user.tenantId, userId: user.id })
   319→    })
   320→    queryClient.invalidateQueries({ queryKey: ['notifications'] })

   829→    queryFn: () => {

   833→    },
   834→    enabled: !!user?.tenantId
   835→  })
   836→
   837→  const mutation = useMutation({
   838→    mutationFn: (body: Record<string, string>) =>

   840→    onSuccess: () => { qc.invalidateQueries({ queryKey: ['documents'] }); toast.success('Document ajouté'); setOpen(false) },
   841→    onError: () => toast.error('Erreur')
   842→  })
   843→
   844→  const deleteMutation = useMutation({
   845→    mutationFn: (id: string) => fetch(`/api/documents/${id}`, { method: 'DELETE' }),
   846→    onSuccess: () => { qc.invalidateQueries({ queryKey: ['documents'] }); toast.success('Document supprimé') },
   847→    onError: () => toast.error('Erreur')
   848→  })
   849→
   850→  const filtered = caseFilter === 'all' ? documents : documents?.filter(d => d.caseId === caseFilter)
   851→

   860→        map.set(key, { clientName, cases: [] })

   862→      const entry = map.get(key)!
   863→      let caseEntry = entry.cases.find(c => c.caseName === caseName)
   864→      if (!caseEntry) { caseEntry = { caseName, docs: [] }; entry.cases.push(caseEntry) }

   866→    })

   868→  }, [filtered])

   877→          </div>
   878→          <Select value={caseFilter} onValueChange={setCaseFilter}><SelectTrigger className="w-full sm:w-48"><SelectValue placeholder="Filtrer par dossier" /></SelectTrigger>

   880→              {cases?.map(c => <SelectItem key={c.id} value={c.id}>{c.reference} - {c.title}</SelectItem>)}</SelectContent></Select>
   881→        </div>
   882→        <Button onClick={() => { setForm({ name: '', description: '', caseId: '' }); setOpen(true) }} className="bg-amber-600 hover:bg-amber-700 text-white shrink-0"><Plus className="size-4" />Ajouter un document</Button>
   883→      </div>
   884→
   885→      {isLoading ? <div className="space-y-2">{[...Array(5)].map((_, i) => <Skeleton key={i} className="h-16" />)}</div> : (
   886→        <div className="space-y-6">

  1030→      {selectedDay && (

  1196→  const messagesEndRef = useRef<HTMLDivElement>(null)
  1197→
  1198→  const { data: messages, isLoading } = useQuery<Message[]>({
  1199→    queryKey: ['messages', user?.tenantId, user?.id],
  1200→    queryFn: () => fetch(`/api/messages?tenantId=${user!.tenantId}&userId=${user!.id}`).then(r => r.json()),
  1201→    enabled: !!user?.tenantId,
  1202→    refetchInterval: 5000
  1203→  })
  1204→
  1205→  const sendMutation = useMutation({
  1206→    mutationFn: (content: string) => fetch('/api/messages', {
  1207→      method: 'POST', headers: { 'Content-Type': 'application/json' },
  1208→      body: JSON.stringify({ content, senderId: user!.id, receiverId: selectedContact, tenantId: user!.tenantId })
  1209→    }).then(r => r.json()),
  1210→    onSuccess: () => { qc.invalidateQueries({ queryKey: ['messages'] }); setMessageText('') },
  1211→    onError: () => toast.error('Erreur d\'envoi')
  1212→  })
  1213→
  1214→  // Build contacts list (unique users from messages)
  1215→  const contacts = useMemo(() => {
  1216→    const map = new Map<string, { id: string; name: string; lastMessage: string; lastTime: string; unread: number }>()
  1217→    messages?.forEach(m => {
  1218→      const other = m.senderId === user?.id ? m.receiver : m.sender
  1219→      if (!other) return
  1220→      const existing = map.get(other.id)
  1221→      if (!existing || new Date(m.createdAt) > new Date(existing.lastTime)) {
  1222→        map.set(other.id, { id: other.id, name: other.name, lastMessage: m.content, lastTime: m.createdAt, unread: existing?.unread || 0 })
  1223→      }
  1224→      if (m.receiverId === user?.id && !m.isRead && m.senderId !== selectedContact) {
  1225→        const e = map.get(other.id)
  1226→        if (e) e.unread++
  1227→      }
  1228→    })
  1229→    return Array.from(map.values()).sort((a, b) => new Date(b.lastTime).getTime() - new Date(a.lastTime).getTime())
  1230→  }, [messages, user?.id, selectedContact])
  1231→
  1232→  const conversation = useMemo(() => {
  1233→    if (!selectedContact || !messages) return []
  1234→    return messages.filter(m =>
  1235→      (m.senderId === selectedContact && m.receiverId === user?.id) ||
  1236→      (m.receiverId === selectedContact && m.senderId === user?.id)
  1237→    ).sort((a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime())
  1238→  }, [messages, selectedContact, user?.id])
  1239→
  1240→  const selectedContactName = contacts.find(c => c.id === selectedContact)?.name
  1241→

  1247→      fetch(`/api/messages/${m.id}`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ isRead: true }) })
  1248→    })
  1249→    if (unread.length > 0) qc.invalidateQueries({ queryKey: ['messages'] })
  1250→  }, [selectedContact, messages, user?.id, qc])
  1251→
  1252→  // Scroll to bottom
  1253→  useEffect(() => {
  1254→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' })
  1255→  }, [conversation.length])
  1256→
  1257→  const handleSend = () => {
  1258→    if (!messageText.trim() || !selectedContact) return
  1259→    sendMutation.mutate(messageText.trim())
  1260→  }
  1261→
  1262→  return (
  1263→    <div className="h-[calc(100vh-8rem)] flex border border-slate-200 rounded-lg overflow-hidden">
  1264→      {/* Contacts panel */}
  1265→      <div className={cn('w-full md:w-80 border-r border-slate-200 flex flex-col bg-white',
  1266→        !showContacts && 'hidden md:flex'
  1267→      )}>
  1268→        <div className="p-4 border-b border-slate-100">
  1269→          <div className="relative">
  1270→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-slate-400" />
  1271→            <Input placeholder="Contacts..." className="pl-9 h-9" />
  1272→          </div>
  1273→        </div>
  1274→        <ScrollArea className="flex-1">
  1275→          {contacts.map(c => (
  1276→            <button key={c.id} onClick={() => { setSelectedContact(c.id); setShowContacts(false) }}
  1277→              className={cn('w-full flex items-center gap-3 p-4 text-left hover:bg-slate-50 transition-colors border-b border-slate-50',
  1278→                selectedContact === c.id && 'bg-amber-50'

  1280→              <Avatar className="size-10 shrink-0"><AvatarFallback className="bg-slate-200 text-slate-600 text-sm">{initials(c.name)}</AvatarFallback></Avatar>

  1366→      map.set(month, (map.get(month) || 0) + i.amount)
  1367→    })
  1368→    return Array.from(map.entries()).map(([name, value]) => ({ name, value }))

  1371→  const topClients = useMemo(() => {

  1377→      existing.total += i.amount
  1378→      existing.count++
  1379→      map.set(name, existing)
  1380→    })
  1381→    return Array.from(map.values()).sort((a, b) => b.total - a.total).slice(0, 10)
  1382→  }, [invoices])
  1383→
  1384→  const kpis = [
  1385→    { label: 'Total revenus', value: fmtMoney(stats.totalRevenue), icon: DollarSign, color: 'text-emerald-600', bg: 'bg-emerald-50' },

  1389→  ]
  1390→
  1391→  return (

  1396→          return <Card key={i}><CardContent className="p-4 flex items-center gap-4">
  1397→            <div className={cn('p-3 rounded-lg', kpi.bg)}><Icon className={cn('size-6', kpi.color)} /></div>
  1398→            <div><p className="text-2xl font-bold text-slate-900">{kpi.value}</p><p className="text-sm text-slate-500">{kpi.label}</p></div>
  1399→          </CardContent></Card>
  1400→        })}
  1401→      </div>
  1402→      <Card><CardHeader><CardTitle className="text-base">Revenus par mois</CardTitle></CardHeader>
  1403→        <CardContent><div className="h-64"><ResponsiveContainer width="100%" height="100%">
  1404→          <BarChart data={monthlyData}><CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" />
  1405→            <XAxis dataKey="name" tick={{ fontSize: 11 }} />
  1406→            <YAxis tick={{ fontSize: 11 }} />

  1411→      </Card>
  1412→      <Card><CardHeader><CardTitle className="text-base">Top clients par facturation</CardTitle></CardHeader>
