   211	function fmtFileSize(bytes: number) {
   212	  if (bytes < 1024) return bytes + ' o'
   213	  if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' Ko'
   214	  return (bytes / 1048576).toFixed(1) + ' Mo'

   250	  const [email, setEmail] = useState('')

   252	  const [loading, setLoading] = useState(false)
   253	
   254	  const handleSubmit = async (e: React.FormEvent) => {
   255	    e.preventDefault()
   256	    if (!email || !password) { toast.error('Veuillez remplir tous les champs'); return }
   257	    setLoading(true)
   258	    try {
   259	      const res = await fetch('/api/auth/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email, password }) })
   260	      const data = await res.json()
   261	      if (!res.ok) { toast.error(data.error || 'Erreur de connexion'); return }
   262	      login(data); toast.success(`Bienvenue, ${data.name} !`)
   263	    } catch { toast.error('Erreur de connexion au serveur') } finally { setLoading(false) }
   264	  }
   265	
   266	  return (

  1017	function CalendarView() {
  1018	  const { user } = useAppStore()
  1019	  const [currentMonth, setCurrentMonth] = useState(new Date())
  1020	  const monthStr = format(currentMonth, 'yyyy-MM')
  1021	
  1022	  const { data: events, isLoading } = useQuery({
  1023	    queryKey: ['events', user?.tenantId, monthStr],
  1024	    queryFn: () => fetch(`/api/events?tenantId=${user?.tenantId}&month=${monthStr}`).then(r => r.json()),
  1025	  })
  1026	
  1027	  const days = useMemo(() => {
  1028	    const monthStart = startOfMonth(currentMonth)
  1029	    const monthEnd = endOfMonth(currentMonth)
  1030	    const calStart = startOfWeek(monthStart, { weekStartsOn: 1 })
  1031	    const calEnd = endOfWeek(monthEnd, { weekStartsOn: 1 })
  1032	    return eachDayOfInterval({ start: calStart, end: calEnd })
  1033	  }, [currentMonth])
  1034	
  1035	  const weekDays = ['Lun', 'Mar', 'Mer', 'Jeu', 'Ven', 'Sam', 'Dim']
  1036	  const getEventsForDay = (day: Date) => (events || []).filter((e: EventItem) => isSameDay(parseISO(e.startTime), day))
  1037	
  1038	  return (
  1039	    <div className="p-4 md:p-6 space-y-4">
  1040	      <div className="flex items-center justify-between">
  1041	        <h2 className="text-lg font-semibold">Calendrier</h2>
  1042	        <div className="flex items-center gap-2">
  1043	          <Button variant="outline" size="icon" className="size-8" onClick={() => setCurrentMonth(subMonths(currentMonth, 1))}><ChevronLeft className="size-4" /></Button>
  1044	          <span className="text-sm font-medium min-w-[140px] text-center">{format(currentMonth, 'MMMM yyyy', { locale: fr })}</span>
  1045	          <Button variant="outline" size="icon" className="size-8" onClick={() => setCurrentMonth(addMonths(currentMonth, 1))}><ChevronRight className="size-4" /></Button>
  1046	        </div>
  1047	      </div>
  1048	
  1049	      {isLoading ? <div className="flex justify-center py-12"><Skeleton className="h-6 w-48" /></div> : (
  1050	        <Card><CardContent className="p-2">
  1051	          <div className="grid grid-cols-7 gap-px bg-slate-100 dark:bg-slate-800 rounded-lg overflow-hidden">
  1052	            {weekDays.map(d => <div key={d} className="bg-white dark:bg-slate-950 p-2 text-center text-xs font-medium text-slate-500 dark:text-slate-400">{d}</div>)}
  1053	            {days.map(day => {
  1054	              const dayEvents = getEventsForDay(day)
  1055	              return (
  1056	                <div key={day.toISOString()} className={cn('bg-white dark:bg-slate-950 p-1 min-h-[80px] md:min-h-[100px]', !isSameMonth(day, currentMonth) && 'opacity-40', isToday(day) && 'bg-amber-50 dark:bg-amber-950/20')}>
  1057	                  <p className={cn('text-xs mb-1', isToday(day) ? 'font-bold text-amber-600' : 'text-slate-500')}>{format(day, 'd')}</p>
