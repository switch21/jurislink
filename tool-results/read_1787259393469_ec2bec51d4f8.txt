   204→  try { return format(parseISO(d), 'dd/MM/yyyy', { locale: fr }) } catch { return '—' }

   206→function fmtDateTime(d: string | null | undefined) {
   207→  if (!d) return '—'

   211→  return new Intl.NumberFormat('fr-FR', { style: 'currency', currency: code, minimumFractionDigits: 0 }).format(amount)

   213→function fmtFileSize(bytes: number) {
   214→  if (bytes < 1024) return bytes + ' o'
   215→  if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' Ko'
   216→  return (bytes / 1048576).toFixed(1) + ' Mo'
   217→}

   261→      const res = await fetch('/api/auth/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email, password }) })

   265→    } catch { toast.error('Erreur de connexion au serveur') } finally { setLoading(false) }

   268→  return (
   269→    <div className="min-h-screen flex items-center justify-center login-pattern p-4 relative overflow-hidden">

   377→            <div className="absolute top-full mt-1 w-full bg-white dark:bg-slate-800 rounded-lg border border-slate-200 dark:border-slate-700 shadow-lg z-50 max-h-80 overflow-y-auto">
