   167→// ==================== Helpers ====================
   168→function fmtDate(d: string | null | undefined) {
   169→  if (!d) return '—'
   170→  try { return format(parseISO(d), 'dd/MM/yyyy', { locale: fr }) } catch { return '—' }
   171→}
   172→function fmtMoney(amount: number, code: string = 'XAF') {
   173→  return new Intl.NumberFormat('fr-FR', { style: 'currency', currency: code, minimumFractionDigits: 0 }).format(amount)
   174→}
   175→function fmtFileSize(bytes: number) {
   176→  if (bytes < 1024) return bytes + ' o'
   177→  if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' Ko'
   178→  return (bytes / 1048576).toFixed(1) + ' Mo'
   179→}
   180→function initials(name: string) {
   181→  return name.split(' ').map(n => n[0]).join('').toUpperCase().slice(0, 2)
   182→}

   224→    e.preventDefault()

   227→    try {
   228→      const res = await fetch('/api/auth/login', {
   229→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   230→        body: JSON.stringify({ email, password })
   231→      })
   232→      const data = await res.json()
   233→      if (!res.ok) { toast.error(data.error || 'Erreur de connexion'); return }

   686→              <TableHead className="hidden lg:table-cell">Email</TableHead><TableHead className="hidden lg:table-cell">Téléphone</TableHead>

   955→  const mutation = useMutation({
   956→    mutationFn: (body: Record<string, string>) =>
   957→      fetch('/api/documents', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ ...body, tenantId: user!.tenantId, userId: user!.id, fileName: body.name, fileType: 'pdf', filePath: '/uploads/' + body.name, fileSize: 0 }) }).then(r => r.json()),

   959→    onError: () => toast.error('Erreur')

   963→    mutationFn: (id: string) => fetch(`/api/documents/${id}`, { method: 'DELETE' }),

   965→    onError: () => toast.error('Erreur')

   969→

   971→    const map = new Map<string, { clientName: string; cases: { caseName: string; docs: Doc[] }[] }>()

   977→      const entry = map.get(key)!
   978→      let caseEntry = entry.cases.find(c => c.caseName === caseName)
   979→      if (!caseEntry) { caseEntry = { caseName, docs: [] }; entry.cases.push(caseEntry) }

   993→          <Select value={caseFilter} onValueChange={setCaseFilter}><SelectTrigger className="w-full sm:w-48"><SelectValue placeholder="Filtrer par dossier" /></SelectTrigger>

   996→        </div>
   997→        <Button onClick={() => { setForm({ name: '', description: '', caseId: '' }); setOpen(true) }} className="bg-amber-600 hover:bg-amber-700 text-white shrink-0"><Plus className="size-4" />Ajouter un document</Button>
   998→      </div>
